local Rayfield = loadstring(game:HttpGet("https://sirius.menu/rayfield"))()

local Window = Rayfield:CreateWindow({
    Name = "CustomHub Scripts Loader",
    Icon = 0,
    LoadingTitle = "CustomHub Loader",
    LoadingSubtitle = "by me",
    ShowText = "CustomHub Loader",
    Theme = "Default",
    ToggleUIKeybind = "K",
    ConfigurationSaving = {
        Enabled = true,
        FolderName = "CustomLoader",
        FileName = "Config"
    }
})

local function SafeLoad(name, url)
    Rayfield:Notify({
        Title = "Loading...",
        Content = "Loading " .. name,
        Duration = 2
    })
    
    task.wait(0.5)
    
    local success, err = pcall(function()
        local source = game:HttpGet(url)
        if not source or #source < 50 then
            error("Empty or corrupted script")
        end
        loadstring(source)()
    end)
    
    if success then
        Rayfield:Notify({
            Title = "✅ Successfully!",
            Content = name .. " Loaded",
            Duration = 4
        })
    else
        Rayfield:Notify({
            Title = "❌ Error",
            Content = name .. ": " .. tostring(err):sub(1, 180),
            Duration = 8
        })
        warn("=== ERROR IN " .. name .. " ===")
        warn(err)
    end
end
